#1attune-tp4-pp2-mt16k2,200.2 tokens/s/GPUTP4 PP2max 16,384 tokens/GPU, no recompute, no offload
  • Starting point: slime's Qwen3-32B script, scripts/run-qwen3-32B.sh with scripts/models/qwen3-32B.sh at slime 0.3.0. slime ships it as a one-node GRPO script that shares the GPUs with its rollout engine and publishes no tokens/s figure for it. Ported to this workload with its layout (TP 8 with sequence parallel, PP 1, full recompute, max 20,480 tokens/GPU, Adam on the CPU) on 8 H200, it measured 1,327.8 tokens/s/GPU; ATTUNE tuned it to 2,200.2 (+66%).
  • No layer is recomputed, so activation memory grows with the tokens per GPU: the tuning runs at max 24,576 tokens per GPU, and at max 16,384 with PP 1 and DP 2, ran out of memory. Check the memory before raising --max-tokens-per-gpu.

export CUDA_DEVICE_MAX_CONNECTIONS=1
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True
export TORCHINDUCTOR_COMPILE_THREADS=1
export NCCL_NVLS_ENABLE=$([ "$NVLINK_COUNT" -gt 0 ] && echo 1 || echo 0)

python3 -m <slime entry module> --async \
  --actor-num-nodes 1 \
  --actor-num-gpus-per-node 8 \
  ${MODEL_ARGS[@]} \
  --hf-checkpoint "$HF_CKPT" \
  --ref-load "$TORCH_DIST" \
  --rollout-function-path pretokenized_sft_rollout.generate_rollout \
  --prompt-data "$SFT_DATA" \
  --input-key messages \
  --metadata-key metadata \
  --num-rollout 20 \
  --rollout-batch-size 128 \
  --global-batch-size 128 \
  --loss-type sft_loss \
  --loss-mask-type qwen3 \
  --calculate-per-token-loss \
  --disable-compute-advantages-and-returns \
  --debug-train-only \
  --tensor-model-parallel-size 4 \
  --sequence-parallel \
  --pipeline-model-parallel-size 2 \
  --context-parallel-size 1 \
  --expert-model-parallel-size 1 \
  --expert-tensor-parallel-size 1 \
  --use-dynamic-batch-size \
  --max-tokens-per-gpu 16384 \
  --optimizer adam --lr 1e-5 --lr-decay-style cosine --min-lr 1e-6 \
  --lr-warmup-fraction 0.1 --weight-decay 0.1 --adam-beta1 0.9 --adam-beta2 0.98 \
  --use-distributed-optimizer \
  --use-precision-aware-optimizer \
  --attention-dropout 0.0 --hidden-dropout 0.0 \
  --accumulate-allreduce-grads-in-fp32 --attention-softmax-in-fp32 \
  --attention-backend flash \
  --distributed-timeout-minutes 30 \
  --tensorboard-dir "$PROFILE_DIR"
